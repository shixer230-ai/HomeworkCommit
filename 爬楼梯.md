//使用组合数数据溢出失败案例
```java
import java.util.Scanner;

public class Test2 {
    public static void main(String[] args) {
        //那就放一个数组存一下吧QwQ
        Scanner sc = new Scanner(System.in);
        System.out.println("请输入测试数");
        int test = sc.nextInt();
        long[] arr = new long[test];
        for (int i = 0; i < test; i++) {
            int stairs = sc.nextInt();
            int temp = stairs;
            long res = 0;
            //计算阶乘
            for (int j = 0; j <= stairs / 2; j++) {
                long com = combination(stairs - j, j);
                res = res + temp * com;
                temp--;
            }
            arr[i] = res;
        }
        //遍历数组
        for (int j = 0; j < arr.length; j++) {
            System.out.println(arr[j]);
        }
    }

    public static long combination(int n, int k) {
        // 参数校验
        if (n < 0 || k < 0 || k > n) {
            throw new IllegalArgumentException("必须满足 0 <= k <= n，当前 n=" + n + ", k=" + k);
        }

        // 利用对称性 C(n, k) = C(n, n-k)，减少计算量
        if (k > n - k) {
            k = n - k;
        }

        // 迭代计算 C(n, k) = (n - k + 1) * (n - k + 2) * ... * n / (1 * 2 * ... * k)
        // 每一步先乘后除，确保中间结果始终为整数
        long result = 1;
        for (int i = 1; i <= k; i++) {
            // 先乘后除，避免使用阶乘导致溢出
            result = result * (n - k + i) / i;
        }
        return result;
    }
}
```
