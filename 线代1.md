import java.util.Scanner;

public class P25 {
    public static void main(String[] args) {
        //这是计算一个逆序数的值的程序
        Scanner sc = new Scanner(System.in);
        System.out.println("请输入一个数以求出其逆序数");
        String str = sc.next();
        System.out.println(getInversion(str));
    }

    public static int getInversion(String str) {
        int count = 0;
        for (int i = 0; i < str.length(); i++) {
            if (str.charAt(i) < '0' || str.charAt(i) > '9') {
                System.out.println("你输入的似乎不是一串数字");
                return -1;
            }
            for (int j = i + 1; j < str.length(); j++) {
                if (str.charAt(i) > str.charAt(j)) {
                    count++;
                }
            }
        }
        System.out.println("该数的逆序数是:");
        return count;
    }
}
